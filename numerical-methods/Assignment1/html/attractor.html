function [] = attractor(min, max, N, iter, buff_points, x0)
%   ATTRACTOR plots Summary of this function goes here
%   Detailed explanation goes here
%   min lambda, max lambda
%   N is the number of lambdas from the interval min-,ax lambda

% default values
if ~exist('N')
    N = 30000;
end

if ~exist('iter')
    iter = 5000;
end

if ~exist('buff_points')
    buff_points = 30;
end

if ~exist('x0')
    x0 = 0.5;
end

% generates N equally-distant lambdas from an interval [min, max]
lambdas = linspace(min, max, N);

% iterates for all lambdas
for i = 1:N
    x = x0;

    % iterates the sequence for a fixed lambda
    for k = 1:iter
        x_n(k) = lambdas(i) * x * (1 - x);
        x = x_n(k);
    end

    % assigns the last approximations of limiting points to a solution set
    % for a fixed lambda
    x_axis(1 + buff_points * (i-1) :  buff_points * i) = ...
        ones(1, buff_points) * lambdas(i);
    y_axis(1 + buff_points * (i-1) :  buff_points * i) = ...
        ones(1,buff_points) .* x_n(iter - buff_points + 1 : iter);
end

% makes a plot
plot(x_axis, y_axis, '.');
xlabel('\lambda');
ylabel('Approximations of cluster points of x_n');
title('Attractor of a sequence x_{n+1}=\lambdax_n(1-x_n)');
dim = [0.2 0.5 0.3 0.3];
str = {sprintf('N = %d', N), sprintf('iterations = %d', iter), ...
    sprintf('buffer points = %d', buff_points), sprintf('x_0 = %.2f', x0)};
annotation('textbox',dim,'String',str,'FitBoxToText','on');

end
